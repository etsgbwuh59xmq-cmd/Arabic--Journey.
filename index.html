App started in English."); // لێرە دەتوانیت دەقەکان بگۆڕیت بۆ ئینگلیزی } else { console.log("بەخێر هاتیت!"); // لێرە دەتوانیت دەقەکان بگۆڕیت بۆ کوردی } } // گۆڕینی ڕەنگی پشتەوە (Background) و پاشەکەوتکردنی bgCircles.forEach(circle => { circle.addEventListener('click', () => { const selectedColor = circle.getAttribute('data-color');
